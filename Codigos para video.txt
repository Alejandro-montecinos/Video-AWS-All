# 🔹 1. PREPARACIÓN DE EC2
# ================================

sudo yum update -y                 
sudo yum upgrade -y                

sudo yum install docker -y         
sudo systemctl enable docker       
sudo systemctl start docker        
sudo usermod -aG docker ec2-user   

sudo yum install git -y            

docker --version                   
git --version                      


# ================================
# 🔹 2. COMANDOS BÁSICOS DOCKER
# ================================

docker images                      # Lista imágenes
docker rmi nombreImagen            # Elimina imagen

docker build -t nombreCarpeta .     # Construye imagen
docker run nombreImagen            # Ejecuta contenedor

docker start nombreContenedor      # Inicia contenedor
docker stop nombreContenedor       # Detiene contenedor
docker restart nombreContenedor    # Reinicia contenedor

docker ps -a                       # Lista contenedores
docker logs nombreContenedor       # Logs del contenedor


# ================================
# 🔹 3. COMANDOS LINUX
# ================================

cd nombreCarpeta                   # Cambia de carpeta
ls                                 # Lista archivos
sudo comando                       # Ejecuta como admin
su - nombreUsuario                 # Cambia usuario
nano archivo.txt                   # Editor de texto
mv archivo.txt {ruta} ejemplo /user/front 
rm -rf directorio               # elimina un directorio sin preguntar ##cuidado con lo que eliminas##

# 🔹 4. VOLUMENES DOCKER

# 🔹 5. AWS CLI CONFIGURACIÓN
# ================================

aws ecr create-repository --repository-name tienda-frontend --region us-east-1   
aws ecr create-repository --repository-name tienda-db --region us-east-1

# ================================
# 🔹 10. DESPLIEGUE MANUAL EN EC2
# ================================

# Copiar proyecto a cada EC2:
/home/ec2-user/frontend   # EC2 WEB
/home/ec2-user/backend    # EC2 BACKEND
/home/ec2-user/db         # EC2 DB


# ----------------
# EC2 DB
# ----------------

cd /home/ec2-user/db
docker build -t tienda-db .   # Construye imagen DB

docker run -d \
--name tienda-db \
-p 3306:3306 \
-v dbdata:/var/lib/mysql \
tienda-db   # Ejecuta MySQL con persistencia


# ----------------
# EC2 BACKEND
# ----------------

cd /home/ec2-user/backend
docker build -t tienda-backend .   # Construye backend

docker run -d \
--name tienda-backend \
-p 3001:3001 \
-e DB_HOST=10.0.2.93 \
-e DB_USER=alumno \
-e DB_PASSWORD=alumno123 \
-e DB_NAME=tienda_perritos \
-e DB_PORT=3306 \
tienda-backend   # Ejecuta backend conectado a DB


# ----------------
# EC2 FRONTEND
# ----------------

cd /home/ec2-user/frontend
docker build -t tienda-frontend .   # Construye frontend

docker run -d \
--name tienda-frontend \
-p 80:80 \
tienda-frontend   # Ejecuta frontend
